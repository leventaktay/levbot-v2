// Felsefe: Swing high/low = likidite havuzu
// Trader'ların %90'ı stop-loss'unu son swing'in
// üstüne/altına koyar → orada likidite birikir
// Smart money bu seviyeleri hedefler
//
// 4H swing high → supply zone (short stop'lar üstte)
// 4H swing low  → demand zone (long stop'lar altta)
// Zone genişliği: pivot ± ATR × zone_width

// Zone dizileri
var box[]   dz_boxes  = array.new_box()
var float[] dz_tops   = array.new_float()
var float[] dz_bots   = array.new_float()
var int[]   dz_state  = array.new_int()      // 0=bekliyor, 1=dokunuldu, 2=sinyal verildi
var int[]   dz_strong = array.new_int()       // 0=normal, 1=1D confluence ★

var box[]   sz_boxes  = array.new_box()
var float[] sz_tops   = array.new_float()
var float[] sz_bots   = array.new_float()
var int[]   sz_state  = array.new_int()
var int[]   sz_strong = array.new_int()

int max_zones = 20
float h4_atr_safe = nz(h4_atr, atr_val)

// --- 1D Confluence kontrolü ---
// Bir 4H pivot, 1D pivot'a yakınsa → güçlendirilmiş zone
f_is_near_d1_hi(price) =>
    d1_confluence and not na(d_last_pvt_hi) and math.abs(price - d_last_pvt_hi) <= h4_atr_safe * confluence_tol

f_is_near_d1_lo(price) =>
    d1_confluence and not na(d_last_pvt_lo) and math.abs(price - d_last_pvt_lo) <= h4_atr_safe * confluence_tol

// --- Yeni 4H Swing High → Supply Zone ---
// Swing high'ın olduğu yerde short stop'lar birikir
// Zone: [high - zone_width×ATR, high + zone_width×ATR]
if mod4_on and not na(h4_pvt_hi)
    float pvt = h4_pvt_hi
    float z_top = pvt + h4_atr_safe * zone_width
    float z_bot = pvt - h4_atr_safe * zone_width

    // Aynı seviyede zaten zone var mı kontrol et (çift çizimi önle)
    bool duplicate = false
    if array.size(sz_tops) > 0
        for j = 0 to array.size(sz_tops) - 1
            if math.abs(array.get(sz_tops, j) - z_top) < h4_atr_safe * 0.2
                duplicate := true
                break

    if not duplicate and z_top > z_bot
        // 1D confluence kontrol
        bool is_strong = f_is_near_d1_hi(pvt)

        color b_col = is_strong ? col_bear : color.new(col_bear, 50)

        if array.size(sz_boxes) >= max_zones
            box.delete(array.shift(sz_boxes))
            array.shift(sz_tops)
            array.shift(sz_bots)
            array.shift(sz_state)
            array.shift(sz_strong)

        nb = box.new(bar_index - sw_4h * 16, z_top, bar_index + 100, z_bot,
             border_color=b_col, bgcolor=z_col, border_width=is_strong ? 2 : 1)
        array.push(sz_boxes, nb)
        array.push(sz_tops, z_top)
        array.push(sz_bots, z_bot)
        array.push(sz_state, 0)
        array.push(sz_strong, is_strong ? 1 : 0)

    float z_bot = pvt - h4_atr_safe * zone_width

    if array.size(dz_tops) > 0
        for j = 0 to array.size(dz_tops) - 1
            if math.abs(array.get(dz_tops, j) - z_top) < h4_atr_safe * 0.2

                break

        color b_col = is_strong ? col_bull : color.new(col_bull, 50)

            array.shift(dz_tops)
